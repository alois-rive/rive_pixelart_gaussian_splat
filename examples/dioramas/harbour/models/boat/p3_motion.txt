# Phase 3, motion: the boat bobs. Parts joined into one mesh (one bone in the blob), a skirt
# below the waterline so a rolled hull never shows its foot, then a seamless 4 s loop at 12 fps:
# roll 5 deg and pitch 2 deg on one period, a 0.035 m rise and fall twice per loop.
# Keys every 2 frames from the curves below, interpolation linear; frame 48 = frame 0 (loop seam).
# hedra exports frame f at f / fps seconds: keys start at frame 0 so the glTF runs 0..4.0 s exactly.
# The export bakes them (`export boat.glb --bake-animations`): the baker samples them at 12 fps.
select all
active Hull
join
rename Hull Boat
active Boat
mode edit
select face where "normal.z < -0.9 && pos.z < 0.01"
extrude 0 0 -0.25
mode object
fps 12
frame range 0 47
select Boat
