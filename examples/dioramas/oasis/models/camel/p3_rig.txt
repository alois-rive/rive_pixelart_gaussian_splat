# Camel, phase 3: rig and walk. Runs after p2_form.txt.
# hedra's `join` maps vertex groups by INDEX, not name: every part first gets the same group list
# (one `parent set --empty-groups` for all of them), its own bone's group is filled, THEN they join.
select all
apply
# one mesh per bone (all verts of a part move rigidly with its bone: a segmented toy)
select Torso Hump Blanket
active Torso
join
select Head Snout EarL EarR
active Head
join
select Tail Tuft
active Tail
join
select LowerFL HoofFL
active LowerFL
join
select LowerFR HoofFR
active LowerFR
join
select LowerBL HoofBL
active LowerBL
join
select LowerBR HoofBR
active LowerBR
join
# the armature, in hedra space (Z up, facing -Y)
add armature --name Rig --location 0 0 0
mode edit
bone select all
bone delete
bone add Body --head 0 0.45 1.0 --tail 0 -0.45 1.0
bone add Neck --head 0 -0.63 0.94 --tail 0 -0.97 1.66 --parent Body
bone add Head --head 0 -0.97 1.66 --tail 0 -1.40 1.72 --parent Neck --connected
bone add Tail --head 0 0.58 1.06 --tail 0 0.78 0.72 --parent Body
bone add ThighFL --head -0.21 -0.35 0.78 --tail -0.21 -0.35 0.42 --parent Body
bone add ShinFL --head -0.21 -0.35 0.42 --tail -0.21 -0.35 0.0 --parent ThighFL --connected
bone add ThighFR --head 0.21 -0.35 0.78 --tail 0.21 -0.35 0.42 --parent Body
bone add ShinFR --head 0.21 -0.35 0.42 --tail 0.21 -0.35 0.0 --parent ThighFR --connected
bone add ThighBL --head -0.21 0.46 0.78 --tail -0.21 0.46 0.42 --parent Body
bone add ShinBL --head -0.21 0.46 0.42 --tail -0.21 0.46 0.0 --parent ThighBL --connected
bone add ThighBR --head 0.21 0.46 0.78 --tail 0.21 0.46 0.42 --parent Body
bone add ShinBR --head 0.21 0.46 0.42 --tail 0.21 0.46 0.0 --parent ThighBR --connected
