# Mine cart, phase 1: blockout of every named part at the spec's sizes. Up Z, facing -Y (the cart
# rolls towards -Y), origin on the ground under the cart's centre. Sizes from cart.spec.md
# (456 px per metre in the front and right panels of art/cart/sheet.v1.png).
new
# the tub: an upside-down frustum, 0.68 x 1.04 at the top (z 0.62), 0.50 x 0.92 at the bottom (z 0.21)
add mesh cube --name Tub --size 1 --location 0 0 0.415
scale 0.68 1.04 0.41
apply --scale
active Tub
mode edit
select face where "normal.z < -0.9"
scale 0.735 0.885 1
mode object
# the iron rim band round the top: a frame of four bars, 0.07 wide (the heap shows inside it)
add mesh cube --name Rim --size 1 --location -0.34 0 0.64
scale 0.07 1.105 0.06
apply --scale
add mesh cube --name RimR --size 1 --location 0.34 0 0.64
scale 0.07 1.105 0.06
apply --scale
add mesh cube --name RimF --size 1 --location 0 -0.5175 0.64
scale 0.75 0.07 0.06
apply --scale
add mesh cube --name RimB --size 1 --location 0 0.5175 0.64
scale 0.75 0.07 0.06

add mesh cylinder --name WheelFR --vertices 8 --radius 0.145 --depth 0.09 --location 0.285 -0.29 0.145 --rotation 0 90 0
