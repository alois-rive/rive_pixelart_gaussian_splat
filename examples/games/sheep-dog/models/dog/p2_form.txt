# Sheepdog, phase 2: form (the bib tapers to a point, the ears lean out, the tail tapers), the eye
# patch and the eyes, DawnBringer 16 materials (sRGB 0..1 of the ramp steps), one object, export.
material new Fur --base-color 0.824 0.490 0.173 --no-slot
material new Cream --base-color 0.871 0.933 0.839 --no-slot
material new Dark --base-color 0.078 0.047 0.110 --no-slot
# the bib: its bottom edge pinched to a point under the chest, as in the front view
active Bib
mode edit
select vert where "pos.z < 0"
scale 0.25 1 1
mode object
# the ears lean out 12 degrees
select EarR
rotate y -12
select EarL
rotate y 12
# the tail narrows towards its tip
active Tail
mode edit
select vert where "pos.y > 0.0"
scale 0.8 0.8 0.8
mode object
# the eyes: dark blocks standing 3 cm proud of the face (a splat would swallow a flush one)
add mesh cube --name EyeR --size 1 --location -0.105 -0.48 0.675
scale 0.07 0.04 0.07
apply --scale
add mesh cube --name EyeL --size 1 --location 0.105 -0.48 0.675
scale 0.07 0.04 0.07
apply --scale
select Body HeadR HeadL EarR EarL LegFR LegFL LegBR LegBL Tail
material assign Fur
select Snout Bib PawFR PawFL PawBR PawBL TailTip
material assign Cream
select Nose EyeR EyeL
material assign Dark
# the patch over the dog's LEFT eye (+X: the image's right in the front view): HeadL's front face
active HeadL
mode edit
select face where "normal.y < -0.9"
material assign Cream
mode object
select all
active Body
join
rename Body Dog
shade flat
export dog.glb
