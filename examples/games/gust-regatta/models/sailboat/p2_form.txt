# Sailboat, phase 2: form and surfacing on top of p1. The cockpit well sunk 0.15 into the deck, the sail's
# pink band, materials in exact mulfok32 colours, the hull parts joined into Hull and the rig into Rig
# (two objects: the game swings and heels the rig as its own bone). Exports sailboat.glb (the gated form).
material new HullRed --base-color 0.9176 0.3843 0.3843 --no-slot
material new Deck --base-color 0.8510 0.7412 0.7843 --no-slot
material new Wood --base-color 0.6510 0.3333 0.3725 --no-slot
material new Spar --base-color 0.2902 0.1882 0.3216 --no-slot
material new Sail --base-color 1 1 1 --no-slot
material new Band --base-color 0.8314 0.4314 0.7020 --no-slot
select Hull
active Hull
material assign HullRed
mode edit
select face where "normal.z > 0.9 && pos.z > 0.45"
bisect --plane-co 0 0.35 0 --plane-no 0 1 0
select face where "normal.z > 0.9 && pos.z > 0.45"
bisect --plane-co 0 1.42 0 --plane-no 0 1 0
select face where "normal.z > 0.9 && pos.z > 0.45"
bisect --plane-co 0.35 0 0 --plane-no 1 0 0
select face where "normal.z > 0.9 && pos.z > 0.45"
bisect --plane-co -0.35 0 0 --plane-no 1 0 0
select face where "normal.z > 0.9 && pos.z > 0.45"
material assign Deck
select face where "normal.z > 0.9 && pos.z > 0.45 && pos.y > 0.35 && pos.y < 1.42 && abs(pos.x) < 0.35"
extrude 0 0 -0.15
material assign Wood
select face where "pos.z > 0.34 && pos.z < 0.5 && pos.y > 0.34 && pos.y < 1.43 && abs(pos.x) < 0.36 && normal.z < 0.5"
material assign Wood
mode object
select Sail
active Sail
material assign Sail
mode edit
select all
bisect --plane-co 0 0 1.27 --plane-no 0 0 1
select all
bisect --plane-co 0 0 1.57 --plane-no 0 0 1
select face where "pos.z > 1.27 && pos.z < 1.57"
material assign Band
mode object
select Mast Boom
material assign Spar
select Rudder Tiller
material assign Wood
select Rudder Tiller Hull
active Hull
join
select Mast Boom Sail
active Mast
join
rename Mast Rig
select all
shade flat
export sailboat.glb
