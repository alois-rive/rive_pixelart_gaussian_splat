# Target, phase 2: form and surfacing. The front cap inset four times into five painted rings of
# equal width (white, night, teal, crimson, gold); straw rim and back; wood legs and runners.
material new Straw --base-color 0.6078 0.6118 0.5098 --no-slot
material new White --base-color 1.0000 1.0000 1.0000 --no-slot
material new Night --base-color 0.2235 0.2039 0.3412 --no-slot
material new Teal --base-color 0.0667 0.6784 0.7569 --no-slot
material new Crimson --base-color 0.7882 0.1412 0.3922 --no-slot
material new Gold --base-color 0.9686 0.8941 0.4627 --no-slot
material new Leg --base-color 0.9765 0.5725 0.3216 --no-slot
material new Runner --base-color 0.7961 0.3020 0.4078 --no-slot
select LegL LegR BackLeg
scale 0.8 0.8 1 --pivot individual --orient local
select RunnerL RunnerR
scale 0.85 1 0.7 --pivot individual --orient local
select TipL TipR
scale 0.85 0.7 1 --pivot individual --orient local
select Boss
scale 1 1 0.9 --orient local
material assign Straw
active Boss
mode edit
select face where "normal.z > 0.9"
material assign White
inset 0.118
material assign Night
inset 0.118
material assign Teal
inset 0.118
material assign Crimson
inset 0.118
material assign Gold
mode object
active RunnerL
mode edit
select face where "normal.z < -0.9"
delete only-faces
mode object
active RunnerR
mode edit
select face where "normal.z < -0.9"
delete only-faces
mode object
select LegL LegR BackLeg
material assign Leg
select RunnerL RunnerR TipL TipR
material assign Runner
select all
shade flat
export target.glb
