# Keeper (Tansy, the greenhouse keeper), phase 1: blockout of every named part at the spec's sizes
# (keeper.spec.md; 560 px per metre in the front panel of art/keeper/sheet.v1.png). Up Z, facing -Y,
# origin on the ground under the body's centre. 0.85 m tall to the leaf's tip.
new
# boots: two chunky blocks, toes forward
add mesh cube --name BootL --size 1 --location 0.085 -0.02 0.05
scale 0.13 0.2 0.1
apply --scale
add mesh cube --name BootR --size 1 --location -0.085 -0.02 0.05
scale 0.13 0.2 0.1
apply --scale
# the barrel body: a low-poly sphere squashed to 0.5 wide, 0.4 deep, 0.36 tall (z 0.08..0.44)
add mesh uv-sphere --name Body --segments 8 --rings 6 --radius 0.25 --location 0 0 0.26
scale 1 0.8 0.72
apply --scale
# the head: 0.34 across, sitting on the body, its top under the hat
add mesh uv-sphere --name Head --segments 8 --rings 6 --radius 0.17 --location 0 -0.01 0.5
scale 1 0.92 0.9
apply --scale
add mesh ico-sphere --name Nose --subdivisions 1 --radius 0.05 --location 0 -0.175 0.49
# the hat: a sloped brim 0.62 across, a crown 0.27 across and 0.14 tall, a ribbon round the crown
add mesh cone --name Brim --vertices 12 --radius1 0.315 --radius2 0.135 --depth 0.1 --location 0 0 0.575
add mesh cone --name Crown --vertices 10 --radius1 0.14 --radius2 0.118 --depth 0.13 --location 0 0 0.675
add mesh cylinder --name Ribbon --vertices 10 --radius 0.144 --depth 0.045 --location 0 0 0.645
# the leaf sprout on the crown (exaggerated: 0.05 thick so it bakes to more than a few splats)
add mesh ico-sphere --name Leaf --subdivisions 1 --radius 0.055 --location 0.012 0 0.785
scale 1.0 0.45 1.3
apply --scale
add mesh cylinder --name Stem --vertices 5 --radius 0.028 --depth 0.06 --location 0 0 0.75
# arms: short stubby cylinders hanging at the sides, mitten hands
add mesh cylinder --name ArmL --vertices 6 --radius 0.06 --depth 0.2 --location 0.215 -0.01 0.29 --rotation 0 -12 0
add mesh cylinder --name ArmR --vertices 6 --radius 0.06 --depth 0.2 --location -0.215 -0.01 0.29 --rotation 0 12 0
select ArmL ArmR
apply --rotation
add mesh ico-sphere --name HandL --subdivisions 1 --radius 0.065 --location 0.24 -0.02 0.175
add mesh ico-sphere --name HandR --subdivisions 1 --radius 0.065 --location -0.24 -0.02 0.175
# the watering can in the LEFT hand (+X: the image's right in the front view), spout forward (-Y)
add mesh cube --name Can --size 1 --location 0.285 -0.03 0.1
scale 0.11 0.15 0.12
apply --scale
add mesh cube --name CanGrip --size 1 --location 0.285 -0.03 0.185
scale 0.04 0.11 0.05
apply --scale
add mesh cylinder --name Spout --vertices 6 --radius 0.022 --depth 0.13 --location 0.285 -0.15 0.14 --rotation -60 0 0
