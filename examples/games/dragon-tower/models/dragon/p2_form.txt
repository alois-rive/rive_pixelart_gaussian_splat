# Dragon, phase 2: materials in exact Blessing colours (sRGB 0..1), joined into the two game parts
# (Body: everything below the neck, Head: head, snout, eyes, nostrils, horns), flat shading, export.
# Scale #96fbc7 (scale ramp step 1), Blush #ffb3cb (blush ramp step 1), Horn #f7ffae, Eye #74569b.
material new Scale --base-color 0.588 0.984 0.780 --no-slot
material new Blush --base-color 1.0 0.702 0.796 --no-slot
material new Horn --base-color 0.969 1.0 0.682 --no-slot
material new Eye --base-color 0.455 0.337 0.608 --no-slot
select Body Seat Back LegL LegR FootL FootR ArmL ArmR WingArmL WingArmR Head Snout Tail0 Tail1 Tail2 Tail3
material assign Scale
select Belly WingL WingR
material assign Blush
select ToeL0 ToeL1 ToeL2 ToeR0 ToeR1 ToeR2 ClawL0 ClawL1 ClawR0 ClawR1 HornL HornR Spade Spike0 Spike1 Spike2 WingClawL0 WingClawL1 WingClawL2 WingClawR0 WingClawR1 WingClawR2
material assign Horn
select EyeL EyeR NostrilL NostrilR
material assign Eye
select Head Snout EyeL EyeR NostrilL NostrilR HornL HornR
active Head
join
select Body Seat Back LegL LegR FootL FootR ArmL ArmR WingArmL WingArmR Tail0 Tail1 Tail2 Tail3 Belly WingL WingR ToeL0 ToeL1 ToeL2 ToeR0 ToeR1 ToeR2 ClawL0 ClawL1 ClawR0 ClawR1 Spade Spike0 Spike1 Spike2 WingClawL0 WingClawL1 WingClawL2 WingClawR0 WingClawR1 WingClawR2
active Body
join
# the head turns about the neck (0, -0.05, 1.3)
cursor set 0 -0.05 1.3
select Head
origin set --to cursor
select all
shade flat
inspect stats
export dragon.glb
