# Bell, phase 2: materials in Citrink's colours (bronze #fcf660, teal #166e7a), hidden faces deleted
# (the open mouth, caps buried inside the body or the dome), flat shading, export. Runs on p1.
material new Bronze --base-color 0.9882 0.9647 0.3765 --no-slot
material new Band --base-color 0.0863 0.4314 0.4784 --no-slot
select Flare Body Dome
material assign Bronze
select BandLow BandHigh Loop BossF BossB
material assign Band
select Flare
active Flare
mode edit
select face where "normal.z > 0.9 || normal.z < -0.9"
delete only-faces
mode object
select Body
active Body
mode edit
select face where "normal.z > 0.9 || normal.z < -0.9"
delete only-faces
mode object
select BandLow
active BandLow
mode edit
select face where "normal.z > 0.9 || normal.z < -0.9"
delete only-faces
mode object
select BandHigh
active BandHigh
mode edit
select face where "normal.z > 0.9 || normal.z < -0.9"
delete only-faces
mode object
select Dome
active Dome
mode edit
select face where "normal.z < -0.9"
delete only-faces
mode object
select BossF
active BossF
mode edit
select face where "normal.y > 0.9"
delete only-faces
mode object
select BossB
active BossB
mode edit
select face where "normal.y < -0.9"
delete only-faces
mode object
select Flare Body BandLow BandHigh Dome Loop BossF BossB
active Body
join
rename Body Bell
shade flat
export bell.glb
