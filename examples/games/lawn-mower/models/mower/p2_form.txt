# Mower, phase 2: form (the hood slopes down to the nose, hubs inset into the wheels), flat shading,
# hidden caps removed, export.
select Hood
active Hood
mode edit
select vert where "pos.y < 0 && pos.z > 0"
translate 0 0 -0.15
mode object
select RearL
active RearL
mode edit
select face where "abs(normal.z) > 0.9"
inset 0.11
material assign Hub
mode object
select RearR
active RearR
mode edit
select face where "abs(normal.z) > 0.9"
inset 0.11
material assign Hub
mode object
select FrontL
active FrontL
mode edit
select face where "abs(normal.z) > 0.9"
inset 0.07
material assign Hub
mode object
select FrontR
active FrontR
mode edit
select face where "abs(normal.z) > 0.9"
inset 0.07
material assign Hub
mode object
select all
shade flat
export mower.glb
